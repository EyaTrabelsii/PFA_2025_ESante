% Executive Summary
\section{Executive Summary}

The E-Santé platform is a comprehensive healthcare management system connecting patients, general practitioners (GPs), and specialists. The system facilitates appointment booking, medical record management, secure messaging, referrals, and multi-channel notifications.

\subsection{Key Features}

\begin{itemize}[leftmargin=*]
    \item \faUsers\ Multi-role authentication (Patients, GPs, Specialists)
    \item \faCalendarAlt\ Appointment scheduling and management
    \item \faCogs\ \textbf{n8n Workflow Automation} for intelligent appointment booking
    \item \faFileMedical\ Electronic medical records (consultations, prescriptions, documents)
    \item \faExchangeAlt\ Referral system between GPs and specialists
    \item \faComments\ Real-time secure messaging
    \item \faBell\ Multi-channel notifications (Push, Email, In-app)
    \item \faClipboardList\ Comprehensive audit logging
    \item \faMapMarkerAlt\ Geolocation-based doctor search
    \item \faMobileAlt\ \textbf{Multi-channel booking} (Web, Mobile, Chatbot, Voice Assistant)
\end{itemize}

\subsection{Technology Stack}

\begin{infobox}[Core Technologies]
\textbf{Backend:} Node.js, Express.js, MongoDB, Kafka, Redis\\
\textbf{Authentication:} JWT, bcrypt\\
\textbf{Real-time:} Socket.IO\\
\textbf{Automation:} n8n workflow engine\\
\textbf{Notifications:} OneSignal (Push), Nodemailer (Email)\\
\textbf{Storage:} AWS S3\\
\textbf{Containerization:} Docker, Docker Compose
\end{infobox}

\subsection{Featured User Story: n8n Workflow Automation}

\subsubsection{US-11.6: n8n Workflow Automation - Appointment Booking}
\begin{tcolorbox}[colback=yellow!10,colframe=warningorange,title=\faLightbulb\ Innovation Feature]
\textbf{As a} patient\\
\textbf{I want} an automated workflow to check doctor availability and book appointments\\
\textbf{So that} I can schedule appointments through multiple channels (chatbot, voice, external integrations)
\end{tcolorbox}

\textbf{Acceptance Criteria:}
\begin{itemize}[label=\textcolor{successgreen}{\faCheck}]
    \item n8n workflow engine integrated with E-Santé API
    \item Workflow: Fetch doctor availability by specialty and location
    \item Workflow: Filter available time slots based on patient preferences
    \item Workflow: Automatically book appointment in available slot
    \item Workflow: Send confirmation via multiple channels (SMS, Email, WhatsApp)
    \item Error handling for unavailable slots or booking failures
    \item Webhook triggers for external systems integration
    \item Support for voice assistant integration (Alexa, Google Assistant)
    \item Chatbot integration (Telegram, WhatsApp, Facebook Messenger)
    \item Appointment reminder workflows (24h, 1h before)
    \item Rescheduling workflow based on doctor cancellations
    \item Waitlist automation (notify patients when slot becomes available)
\end{itemize}

\textbf{Story Points:} 13 \hfill \textbf{Status:} \textcolor{successgreen}{\faCheckCircle\ COMPLETED}

\subsubsection{Technical Implementation}

\textbf{n8n workflow connects to E-Santé REST API:}

\begin{itemize}
    \item \texttt{GET /api/v1/users/doctors/search}
    \item \texttt{GET /api/v1/rdv/availability}
    \item \texttt{POST /api/v1/rdv/appointments}
\end{itemize}

\textbf{Workflow Steps:}
\begin{enumerate}
    \item \textbf{Trigger:} Webhook/Chatbot/Voice command with patient request
    \item \textbf{Call:} GET doctor availability API
    \item \textbf{Filter:} Available slots matching criteria
    \item \textbf{Call:} POST create appointment
    \item \textbf{Send:} Multi-channel notifications
    \item \textbf{Log:} Audit event
\end{enumerate}

\textbf{Business Value:}
\begin{itemize}
    \item Reduces manual booking effort for patients
    \item Enables 24/7 automated appointment booking
    \item Improves accessibility through multiple channels
    \item Reduces reception workload for clinics
    \item Provides better patient experience
\end{itemize}

\subsubsection{Example Use Cases}

\begin{successbox}[Use Case Example]

    \item Shows 3 available slots this week

\textbf{Patient receives:} WhatsApp confirmation + Email + Push notification
